const fieldEventNames={issue_technician_stock:'Kho chuẩn bị & cấp hàng',receive_technician_stock:'Nhận hàng mang theo',consume_technician_stock:'Ghi hàng đã sử dụng',request_stock_return:'Đề nghị trả hàng dư',confirm_stock_return:'Kho xác nhận nhận lại',report_payment:'Báo khoản thu khách',confirm_payment:'Finance xác nhận khoản thu',request_cash_handover:'Đề nghị nộp quỹ',confirm_cash_handover:'Quỹ xác nhận nhận tiền'};
Object.assign(eventNames,fieldEventNames);
Object.assign(names,{reported:'Chờ xác nhận thu',partial:'Đã thu một phần',paid:'Đã thu đủ'});
Object.assign(errors,{INVALID_QUANTITY_OR_AMOUNT:'Số lượng hoặc số tiền phải là số nguyên dương.',UNKNOWN_WAREHOUSE_BALANCE:'Chưa có số dư kho mẫu cho sản phẩm này.',STOCK_PICKUP_REQUIRED:'Cần kho cấp đủ hàng và kỹ thuật xác nhận nhận hàng trước khi bắt đầu tới khách.',INSUFFICIENT_WAREHOUSE_STOCK:'Kho mẫu không đủ hàng.',INSUFFICIENT_TECHNICIAN_STOCK:'Hàng giữ không đủ hoặc đã được đề nghị trả.',ORDER_QUANTITY_EXCEEDED:'Số cấp vượt số lượng trên dòng đơn.',PAYMENT_EXCEEDS_ORDER:'Khoản thu vượt giá trị đơn sau khi tính cả các khoản đang chờ xác nhận.',DUPLICATE_PAYMENT_REFERENCE:'Mã biên nhận hoặc giao dịch này đã được dùng.',HANDOVER_EXCEEDS_CASH:'Số nộp vượt khoản tiền đã xác nhận còn có thể nộp.',CONFIRMED_CASH_REQUIRED:'Chỉ nộp quỹ từ khoản tiền mặt đã được Finance xác nhận.',DUPLICATE_HANDOVER_REFERENCE:'Mã đề nghị nộp quỹ này đã được dùng.'});
const fieldLedger=()=>state.field_operations||{payments:{},handovers:{},stock_movements:[],stock_returns:{},warehouse_balances:{}};
const paymentRows=o=>Object.values(fieldLedger().payments).filter(p=>p.order_id===o.id);
function stockPanel(o){
 const r=actor().role,lines=o.lines.filter(l=>masters.products[l.product_id]?.kind==='physical');
 if(!lines.length||!['technician','warehouse','manager'].includes(r))return '';
 const buttons=[];const button=(type)=>buttons.push(`<button data-action="${type}">${e(eventNames[type])}</button>`);
 if(!o.reversal&&!['draft','cancelled','reversal_approved'].includes(o.status)){
  if(r==='warehouse'&&o.assignee_id)button('issue_technician_stock');
  if(r==='technician'&&o.assignee_id===actor().id){if(ErpDomain.pendingPickup(state,o.id))button('receive_technician_stock');if(['doing','testing'].includes(o.work_status))button('consume_technician_stock');button('request_stock_return');}
 }
 const returns=Object.values(fieldLedger().stock_returns).filter(x=>x.order_id===o.id);
 if(r==='warehouse'&&returns.some(x=>x.status==='proposed'))button('confirm_stock_return');
 return `<div class="panel"><div class="panel-head"><h2>Kho thợ · hàng theo đơn</h2></div><div class="panel-body">${lines.map(l=>`<p>${e(l.name)} · Số đơn: ${l.qty} · Chờ kỹ thuật nhận: <strong>${ErpDomain.pendingPickup(state,o.id,l.id)}</strong> · Đã nhận, đang giữ: <strong>${ErpDomain.stockBalance(state,o.id,l.id)}</strong></p>`).join('')}<p class="muted">Kho cấp hàng → kỹ thuật đến kho kiểm đếm và xác nhận nhận mang theo → bắt đầu tới khách. Hàng chờ nhận chưa tính vào hàng kỹ thuật đang giữ. Đề nghị trả giữ chỗ, chưa nhập lại kho.</p>${returns.map(x=>`<p>Đề nghị trả ${x.qty} · ${e(x.reference)} · ${pill(x.status)}</p>`).join('')}<div class="actions">${buttons.join('')}</div></div></div>`;
}
function paymentPanel(o){
 const role=actor().role;if(role==='warehouse')return '';
 const rows=paymentRows(o).filter(p=>role!=='technician'||p.collected_by===actor().id),summary=ErpDomain.paymentSummary(state,o.id),buttons=[];
 const button=type=>buttons.push(`<button data-action="${type}">${e(eventNames[type])}</button>`);
 if(['sales','technician'].includes(role)&&!o.reversal&&!['draft','cancelled','reversal_approved'].includes(o.status))button('report_payment');
 if(role==='finance'&&rows.some(p=>p.status==='reported'))button('confirm_payment');
 if(['sales','technician'].includes(role)&&rows.some(p=>p.method==='cash'&&p.status==='confirmed'&&p.collected_by===actor().id))button('request_cash_handover');
 const handovers=Object.values(fieldLedger().handovers).filter(h=>h.order_id===o.id&&(role!=='technician'||h.collected_by===actor().id));
 if(role==='finance'&&handovers.some(h=>h.status==='proposed'))button('confirm_cash_handover');
 return `<div class="panel"><div class="panel-head"><h2>Thu khách & nộp quỹ · dữ liệu mẫu</h2></div><div class="panel-body">${role!=='technician'?`<p>Đã xác nhận thu: <strong>${money(summary.confirmed_vnd)}</strong> · Còn phải thu: <strong>${money(summary.remaining_vnd)}</strong></p><p>Đang chờ xác nhận: ${money(summary.pending_vnd)}</p>`:''}<p class="muted">Báo thu chưa giảm phải thu. Nộp quỹ chưa xác nhận vẫn tính là tiền đang giữ. Doanh thu được duyệt riêng.</p>${rows.map(p=>`<p>${e(p.reference)} · ${money(p.amount_vnd)} · ${e({cash:'Tiền mặt',bank:'Chuyển khoản',pos:'POS'}[p.method])} · ${pill(p.status)}</p>`).join('')||'<p class="muted">Chưa có khoản thu.</p>'}${handovers.map(h=>`<p>Nộp quỹ ${e(h.reference)} · ${money(h.amount_vnd)} · ${pill(h.status)}</p>`).join('')}<div class="actions">${buttons.join('')}</div></div></div>`;
}

 const ids=['finance','manager'].includes(actor().role)?[...new Set(Object.values(fieldLedger().payments).map(p=>p.collected_by))]:[actor().id];

function fieldForm(type,o){
 const select=(label,id,options)=>`<label for="${id}">${label}</label><select id="${id}">${options.map(([value,title])=>`<option value="${e(value)}">${e(title)}</option>`).join('')}</select>`;
 const number=(label,id)=>`<label for="${id}" style="margin-top:12px">${label}</label><input id="${id}" type="number" min="1" step="1" required>`;
 const evidence=(label='Nội dung xác nhận')=>`<label for="evidence" style="margin-top:12px">${label}</label><textarea id="evidence" required></textarea>`;
 const automatic='<p class="muted">Số phiếu/biên nhận được tự động cấp khi lưu.</p>';
 const note='<label for="evidence" style="margin-top:12px">Ghi chú (không bắt buộc)</label><textarea id="evidence"></textarea>';
 if(['issue_technician_stock','consume_technician_stock','request_stock_return'].includes(type))return select('Dòng hàng vật lý','field-line',o.lines.filter(l=>masters.products[l.product_id]?.kind==='physical').map(l=>[l.id,l.name]))+number('Số lượng','field-qty')+automatic+note;
 if(type==='receive_technician_stock')return select('Phiếu kho cấp đang chờ nhận','field-record',fieldLedger().stock_movements.filter(m=>m.order_id===o.id&&m.type==='issue'&&m.status==='awaiting_receipt'&&m.technician_id===actor().id).map(m=>[m.id,`${m.reference} · ${o.lines.find(l=>l.id===m.line_id)?.name} · SL ${m.qty}`]))+evidence('Xác nhận đã kiểm đếm và lấy hàng mang theo');
 if(type==='confirm_stock_return')return select('Đề nghị trả hàng','field-record',Object.values(fieldLedger().stock_returns).filter(x=>x.order_id===o.id&&x.status==='proposed').map(x=>[x.id,`${x.reference} · ${x.qty}`]))+evidence('Bằng chứng kho nhận lại');
 if(type==='report_payment')return number('Số tiền thu mẫu (VND)','field-amount')+select('Phương thức','field-method',[['cash','Tiền mặt'],['bank','Chuyển khoản'],['pos','POS']])+automatic+'<label for="external-reference">Mã giao dịch ngân hàng/POS (nếu có)</label><input id="external-reference">'+note;
 if(type==='confirm_payment')return select('Khoản thu chờ xác nhận','field-record',paymentRows(o).filter(p=>p.status==='reported').map(p=>[p.id,`${p.reference} · ${money(p.amount_vnd)}`]))+evidence('Căn cứ Finance xác nhận');
 if(type==='request_cash_handover')return select('Khoản tiền mặt của bạn đã xác nhận','field-record',paymentRows(o).filter(p=>p.method==='cash'&&p.status==='confirmed'&&p.collected_by===actor().id).map(p=>[p.id,`${p.reference} · ${money(p.amount_vnd)}`]))+number('Số tiền nộp mẫu (VND)','field-amount')+automatic+note;
 if(type==='confirm_cash_handover')return select('Đề nghị nộp quỹ','field-record',Object.values(fieldLedger().handovers).filter(h=>h.order_id===o.id&&h.status==='proposed').map(h=>[h.id,`${h.reference} · ${money(h.amount_vnd)}`]))+evidence('Bằng chứng quỹ đã nhận tiền');
 return null;
}
function fieldPayload(type){
 const value=id=>document.getElementById(id)?.value;
 if(!Object.hasOwn(fieldEventNames,type))return null;
 const p={order_id:form.order_id},evidence=value('evidence');
 if(['issue_technician_stock','consume_technician_stock','request_stock_return'].includes(type))return {...p,line_id:value('field-line'),qty:Number(value('field-qty')),note:evidence};
 if(type==='report_payment')return {...p,amount_vnd:Number(value('field-amount')),method:value('field-method'),external_reference:value('external-reference'),note:evidence};
 if(type==='request_cash_handover')return {...p,allocations:[{payment_id:value('field-record'),amount_vnd:Number(value('field-amount'))}],note:evidence};
 return {...p,[{receive_technician_stock:'issue_id',confirm_payment:'payment_id',confirm_stock_return:'return_id',confirm_cash_handover:'handover_id'}[type]]:value('field-record'),evidence};
}
