/** Owner-only, COPY-first rollout. No endpoint accepts migration or storage flags. */
function createDispatchTablesCopy(){dispatchBridgeOwner_();var p=dispatchRuntime_();if(p.getProperty('DISPATCH_STORAGE_MODE')==='TABLES')throw new Error('TABLES_ALREADY_ACTIVE');return dispatchLock_(function(){var source=p.getProperty('DISPATCH_SHEET_ID'),file=DriveApp.getFileById(source).makeCopy('DISPATCH_TABLES_COPY_'+Utilities.formatDate(new Date(),'Asia/Ho_Chi_Minh','yyyyMMdd_HHmmss'));file.setSharing(DriveApp.Access.PRIVATE,DriveApp.Permission.NONE);var book=SpreadsheetApp.openById(file.getId());book.insertSheet('DP_COPY_GUARD').getRange(1,1,2,2).setValues([['source_id','token'],[source,p.getProperty('DISPATCH_MARKER_TOKEN')]]);p.setProperty('DISPATCH_TABLES_COPY_ID',book.getId());return {copy_id:book.getId(),active:false};});}
function dispatchTablesCopy_(p){var id=p.getProperty('DISPATCH_TABLES_COPY_ID'),source=p.getProperty('DISPATCH_SHEET_ID');if(!id||id===source)throw new Error('TABLES_COPY_REQUIRED');var book=SpreadsheetApp.openById(id),guard=book.getSheetByName('DP_COPY_GUARD');if(!guard||JSON.stringify(guard.getRange(1,1,2,2).getDisplayValues())!==JSON.stringify([['source_id','token'],[source,p.getProperty('DISPATCH_MARKER_TOKEN')]]))throw new Error('TABLES_COPY_REQUIRED');return book;}
function migrateDispatchToTables(){dispatchBridgeOwner_();var p=dispatchRuntime_(),book=dispatchTablesCopy_(p),legacyBook=SpreadsheetApp.openById(p.getProperty('DISPATCH_SHEET_ID')),legacy=new DispatchBundle.SheetsRepository({workbook:legacyBook,stagingId:legacyBook.getId(),markerToken:p.getProperty('DISPATCH_MARKER_TOKEN'),domain:DispatchBundle.Domain,withLock:dispatchLock_,hash:DispatchBundle.digest,id:function(){return Utilities.getUuid();},flush:function(){SpreadsheetApp.flush();}}),driver=new DispatchBundle.TableDriver.SheetsTableDriver(book,{flush:function(){SpreadsheetApp.flush();}});
 var result=DispatchBundle.TableMigration.migrate({legacy:legacy,driver:driver,withLock:dispatchLock_,hash:DispatchBundle.digest,copyVerified:true,backup:function(snapshot){var name='BACKUP_'+Utilities.formatDate(new Date(),'Asia/Ho_Chi_Minh','yyyyMMdd_HHmmss')+'_'+Utilities.getUuid().slice(0,6),s=book.insertSheet(name),raw=JSON.stringify(snapshot),chunks=[];for(var i=0;i<raw.length;i+=15000)chunks.push([i/15000,DispatchBundle.digest(raw),raw.slice(i,i+15000)]);s.getRange(1,1,1,3).setValues([['chunk','checksum','json']]);if(chunks.length+1>s.getMaxRows())s.insertRowsAfter(s.getMaxRows(),chunks.length+1-s.getMaxRows());s.getRange(2,1,chunks.length,3).setValues(chunks);SpreadsheetApp.flush();var restored=s.getRange(2,3,chunks.length,1).getValues().map(function(r){return r[0];}).join('');if(DispatchBundle.digest(restored)!==DispatchBundle.digest(raw))throw new Error('MIGRATION_BACKUP_MISMATCH');return name;},activate:function(meta){p.setProperty('DISPATCH_STORAGE_MODE','TABLES');dispatchPublishChanges_(p,meta.revision);}});console.log(JSON.stringify(result));return result;}
function rollbackDispatchToSnapshot(){dispatchBridgeOwner_();var p=dispatchRuntime_(),book=dispatchTablesCopy_(p),driver=new DispatchBundle.TableDriver.SheetsTableDriver(book,{flush:function(){SpreadsheetApp.flush();}});return DispatchBundle.TableMigration.rollback({driver:driver,withLock:dispatchLock_,copyVerified:true,hash:DispatchBundle.digest,legacy:new DispatchBundle.SheetsRepository({workbook:SpreadsheetApp.openById(p.getProperty('DISPATCH_SHEET_ID')),stagingId:p.getProperty('DISPATCH_SHEET_ID'),markerToken:p.getProperty('DISPATCH_MARKER_TOKEN'),domain:DispatchBundle.Domain,withLock:dispatchLock_,hash:DispatchBundle.digest,id:function(){return Utilities.getUuid();},flush:function(){SpreadsheetApp.flush();}}),deactivate:function(meta){p.setProperty('DISPATCH_STORAGE_MODE','LEGACY');dispatchPublishChanges_(p,meta.revision);}});}
function archiveDispatchClosedOrders(){dispatchBridgeOwner_();var p=dispatchRuntime_();if(p.getProperty('DISPATCH_STORAGE_MODE')!=='TABLES')throw new Error('TABLES_NOT_ACTIVE');var book=dispatchTablesCopy_(p),repo=new DispatchBundle.Tables.TableRepository({driver:new DispatchBundle.TableDriver.SheetsTableDriver(book,{flush:function(){SpreadsheetApp.flush();}}),withLock:dispatchLock_,hash:DispatchBundle.digest,id:function(){return Utilities.getUuid();}});var result=repo.archiveClosed(new Date(Date.now()-30*86400000).toISOString());console.log(JSON.stringify(result));return result;}
/** Read-only numeric profile. Run on the verified copy; never logs customer fields. */
function profileDispatchTables(){dispatchBridgeOwner_();var p=dispatchRuntime_(),book=dispatchTablesCopy_(p),driver=new DispatchBundle.TableDriver.SheetsTableDriver(book,{flush:function(){SpreadsheetApp.flush();}}),repo=new DispatchBundle.Tables.TableRepository({driver:driver,withLock:dispatchLock_,hash:DispatchBundle.digest,id:function(){return Utilities.getUuid();}}),start=Date.now(),snapshot=repo.snapshot(),result={measured:true,storage_mode:p.getProperty('DISPATCH_STORAGE_MODE')||'LEGACY',revision:snapshot.revision,list_ms:Date.now()-start,open_orders:Object.keys(snapshot.state.orders).length,rows:driver.counts(),allocated_cells:book.getSheets().reduce(function(n,s){return n+s.getMaxRows()*s.getMaxColumns();},0),checked_at:new Date().toISOString()};console.log(JSON.stringify(result));return result;}
function installDispatchTablesArchive(){dispatchBridgeOwner_();var p=dispatchRuntime_();dispatchTablesCopy_(p);if(p.getProperty('DISPATCH_STORAGE_MODE')!=='TABLES')throw new Error('TABLES_NOT_ACTIVE');ScriptApp.getProjectTriggers().filter(function(t){return t.getHandlerFunction()==='dispatchTablesArchiveTick';}).forEach(function(t){ScriptApp.deleteTrigger(t);});ScriptApp.newTrigger('dispatchTablesArchiveTick').timeBased().everyDays(1).atHour(2).inTimezone('Asia/Ho_Chi_Minh').create();return {installed:true};}
