var DISPATCH_CONSUMER_TARGET='finance',DISPATCH_FINANCE_INTERNAL_CONSUME_=false;
function dispatchBridgeConsume_(j){var user=getCurrentUser_();if(!['ADMIN','KE_TOAN'].includes(user.role))throw new Error('FORBIDDEN');var previous=DISPATCH_FINANCE_INTERNAL_CONSUME_;try{DISPATCH_FINANCE_INTERNAL_CONSUME_=true;return dispatchBridgeConsumeFinance_(j);}finally{DISPATCH_FINANCE_INTERNAL_CONSUME_=previous;}}
function dispatchBridgeGuard_(){dispatchBridgeBook_('finance');if(PropertiesService.getScriptProperties().getProperty('DISPATCH_BRIDGE_ENABLED')!=='true')throw new Error('INTEGRATION_DISABLED');var u=getCurrentUser_();if(['ADMIN','KE_TOAN'].indexOf(u.role)<0)throw new Error('FORBIDDEN');}
function dispatchFindExternalFinance_(key,kind){var table=kind==='THU'?SHEETS.TRANSACTIONS:kind==='CHUYEN_QUY'?SHEETS.TRANSFERS:SHEETS.CUSTOMER_DEBTS,marker='['+key+']',found=readObjects_(table).filter(function(r){return String(r.GHI_CHU||'').indexOf(marker)>=0;});if(found.length>1)throw new Error('INTEGRATION_DUPLICATE_EXTERNAL_KEY');if(!found.length)return null;return {ok:true,code:found[0].MA_PHIEU||found[0].HOA_DON_ID,external_key:key,replayed:true};}
/** Durable cancellation fence: a late original THU cannot race past an approved correction. */
function dispatchFinanceCorrectionRows_(){var sh=dispatchBridgeBook_('finance').getSheetByName('DISPATCH_FINANCE_CORRECTIONS');if(!sh)return [];return sh.getLastRow()>1?sh.getRange(2,1,sh.getLastRow()-1,5).getValues():[];}
function dispatchFinanceCorrectionFingerprint_(j){return JSON.stringify({key:j.key,source_key:j.source_key,source_date:j.source_date,correction_reason:j.correction_reason,source_lines:j.source_lines,mapping_snapshot:j.mapping_snapshot});}
function dispatchAssertFinanceExternal_(payload,replayCheckOnly){var key=String(payload.externalKey||'');if(!key)return;if(/^DIEUPHOI:/.test(key)&&!DISPATCH_FINANCE_INTERNAL_CONSUME_)throw new Error('FORBIDDEN');var rows=dispatchFinanceCorrectionRows_();if(rows.some(function(r){return key===r[1]||key.indexOf(r[1]+':')===0;}))throw new Error('FINANCE_SOURCE_REPLACED');if(replayCheckOnly)return;if(payload.correctionCancelKey){if(!rows.some(function(r){return r[0]===payload.correctionCancelKey&&r[2]==='DONE';}))throw new Error('FINANCE_CANCELLATION_NOT_COMPLETE');if(isDateClosed_(payload.correctionSourceDate))throw new Error('FINANCE_DATE_CLOSED: ngày PT gốc đã chốt, cần Admin mở khóa');}}
function dispatchVoidCash_(j){
 var lock=LockService.getScriptLock();lock.waitLock(30000);try{
  var book=dispatchBridgeBook_('finance'),sheet=book.getSheetByName('DISPATCH_FINANCE_CORRECTIONS'),rows=dispatchFinanceCorrectionRows_(),existing=rows.find(function(r){return r[0]===j.key;}),fingerprint=dispatchFinanceCorrectionFingerprint_(j);
  if(existing&&existing[3]!==fingerprint)throw new Error('INTEGRATION_IDEMPOTENCY_CONFLICT');
  if(existing&&existing[2]==='DONE')return JSON.parse(existing[4]);
  if(isDateClosed_(j.source_date))throw new Error('FINANCE_DATE_CLOSED: ngày PT gốc đã chốt, cần Admin mở khóa');
  if(!sheet){sheet=book.insertSheet('DISPATCH_FINANCE_CORRECTIONS');sheet.appendRow(['EXTERNAL_KEY','SOURCE_KEY','STATUS','FINGERPRINT','RESULT']);}
  if(!existing){sheet.appendRow([j.key,j.source_key,'PREPARED',fingerprint,'']);SpreadsheetApp.flush();}
 }finally{lock.releaseLock();}
 var docs=[];j.source_lines.forEach(function(line){var marker='['+j.source_key+':'+(line.external_suffix||line.category)+']',found=readObjects_(SHEETS.TRANSACTIONS).filter(function(r){return String(r.GHI_CHU||'').indexOf(marker)>=0;});if(found.length>1)throw new Error('INTEGRATION_DUPLICATE_EXTERNAL_KEY');if(!found.length)return;var row=found[0],cat=j.mapping_snapshot.categories[line.category];if(row.LOAI!=='THU'||String(row.TAI_KHOAN_ID)!==String(j.mapping_snapshot.account.id)||String(row.DANH_MUC)!==String(cat.code)||Number(row.SO_TIEN)!==line.amount)throw new Error('FINANCE_SOURCE_CHANGED');
  if(row.TRANG_THAI==='DA_HUY'){if(String(row.GHI_CHU||'').indexOf('['+j.key+']')<0)throw new Error('FINANCE_SOURCE_CHANGED');}else deleteTransaction({id:row.ID_GIAO_DICH,reason:'['+j.key+'] '+(j.correction_reason||j.reason)});docs.push({code:row.MA_PHIEU,external_key:j.key,status:'DA_HUY'});
 });
 lock=LockService.getScriptLock();lock.waitLock(30000);try{var all=dispatchFinanceCorrectionRows_(),index=all.findIndex(function(r){return r[0]===j.key;}),result={documents:docs,cancelled:true,original_not_recorded:docs.length===0,external_key:j.key};if(index<0)throw new Error('INTEGRATION_CORRECTION_FENCE_MISSING');dispatchBridgeBook_('finance').getSheetByName('DISPATCH_FINANCE_CORRECTIONS').getRange(index+2,3,1,3).setValues([['DONE',fingerprint,JSON.stringify(result)]]);SpreadsheetApp.flush();return result;}finally{lock.releaseLock();}
}

function dispatchBridgeConsumeFinance_(j){if(j.target!=='finance'||!j.mapping_snapshot)throw new Error('INVALID_FINANCE_MAPPING');var m=j.mapping_snapshot,done=[];if(j.kind==='VOID_ORIGINAL')return {superseded:true};if(j.kind==='CASH_VOID')return dispatchVoidCash_(j);
 if(j.kind==='THU'){j.lines.forEach(function(line){var category=m.categories[line.category];if(!category||!m.account)throw new Error('INVALID_FINANCE_MAPPING');var debtRef='';if(line.debt_key){var debtRows=readObjects_(SHEETS.CUSTOMER_DEBTS).filter(function(r){return String(r.GHI_CHU||'').indexOf('['+line.debt_key+']')>=0;});if(debtRows.length!==1)throw new Error('FINANCE_SOURCE_DEBT_NOT_RECORDED');debtRef=debtRows[0].ID_CONG_NO;}done.push(saveTransaction({...(debtRef?{dispatchDebt:true,allocations:[{debtId:debtRef,amount:line.amount}]}:{}),externalKey:j.key+':'+(line.external_suffix||line.category),...(j.correction_id?{correctionCancelKey:j.dependency_key,correctionSourceDate:j.source_date}:{}),ngayGiaoDich:j.date,loai:'THU',danhMuc:category.code,taiKhoanId:m.account.id,soTien:line.amount,noiDung:j.document_no+' / '+j.order_no,ghiChu:'['+j.key+':'+(line.external_suffix||line.category)+'] '+j.document_no+' / '+j.order_no,phuongThuc:j.payment_kind==='cash'?'TIEN_MAT':'CHUYEN_KHOAN'}));});}
 else if(j.kind==='CHUYEN_QUY'){if(!m.destination||!m.employee||m.sources.length!==j.sources.length)throw new Error('INVALID_FINANCE_MAPPING');j.sources.forEach(function(src,index){var sourceSheet=dispatchBridgeBook_('outbox').getSheetByName('DISPATCH_OUTBOX'),sourceRows=sourceSheet.getRange(2,1,sourceSheet.getLastRow()-1,6).getValues(),sourceRow=sourceRows.find(function(r){return r[0]===src.key;});var prior=false;if(sourceRow&&sourceRow[4]){dispatchBridgeVerify_(sourceRow[4],sourceRow[5]);var report=JSON.parse(sourceRow[4]);prior=report.key===src.key&&report.ok===true;}if(!prior)throw new Error('FINANCE_SOURCE_PT_NOT_RECORDED');done.push(saveInternalTransfer({externalKey:j.key+':'+index,ngayChuyen:j.date,taiKhoanNguonId:m.sources[index].id,taiKhoanDichId:m.destination.id,soTien:src.amount,nhanVienNopId:m.employee.id,noiDung:j.document_no+' / '+src.key,ghiChu:'['+j.key+':'+index+'] '+j.document_no}));});}
 else if(j.kind==='CONG_NO_KH'){if(!m.customer)throw new Error('INVALID_FINANCE_MAPPING');done.push(saveDebt({externalKey:j.key,type:'KH',partyId:m.customer.id,invoiceId:j.document_no,invoiceDate:j.date,dueDate:j.due_date,total:j.amount,content:j.order_no,note:'['+j.key+']',referenceCode:j.order_no}));}else throw new Error('INVALID_SYNC_DOCUMENT');return {documents:done.map(function(r){return {code:r.code||r.invoiceId||r.debtId,external_key:r.external_key};})};}
