/** Proposed AI_KHO patch: never call a stock setter. Runs only on verified copies pending owner's review. */
function aiSubmitToKho_(spreadsheet,draft,plan){var props=PropertiesService.getScriptProperties();if(props.getProperty('AI_KHO_BRIDGE_APPROVED')!=='true')throw new Error('AI_KHO_OWNER_APPROVAL_REQUIRED');DispatchIntegration.verified(SpreadsheetApp.openById.bind(SpreadsheetApp),spreadsheet.getId(),props.getProperty('DISPATCH_COPY_TOKEN'));if(plan.actions.some(function(a){return !['stock_in','stock_out'].includes(a.type);}))throw new Error('KHO_SINGLE_WRITER_UNSUPPORTED_OPERATION');var jobs=[];plan.actions.filter(function(a){return a.type==='stock_in'||a.type==='stock_out';}).forEach(function(a,index){var lines=a.items.map(function(item){if(item.warehouse!=='TONKHO')throw new Error('CHOTHUE_NOT_SUPPORTED_BY_KHO_BRIDGE');var code=String(spreadsheet.getSheetByName(draft.names.stock).getRange(item.row,7).getDisplayValue()||'').trim();if(!/^TD-\d+$/i.test(code))throw new Error('UNKNOWN_TD_CODE');return {code:code,name:item.name,unit:item.unit,qty:item.quantity,serial:item.serial||''};});jobs.push({key:'AI_KHO:'+draft.requestId+':'+index,target:'warehouse',kind:a.type==='stock_in'?'IN':'OUT',warehouse:'145',document_no:draft.requestId,order_no:'AI_KHO',actor_id:a.person,date:draft.date||Utilities.formatDate(new Date(),'Asia/Ho_Chi_Minh','yyyy-MM-dd'),lines:lines,attempt:0,status:'pending'});});var outbox=dispatchBridgeBook_('outbox').getSheetByName('DISPATCH_OUTBOX');if(!outbox)throw new Error('INTEGRATION_OUTBOX_REQUIRED');var existing=outbox.getLastRow()>1?outbox.getRange(2,1,outbox.getLastRow()-1,1).getValues().map(function(r){return r[0];}):[];jobs.forEach(function(j){if(existing.indexOf(j.key)>=0)return;var raw=JSON.stringify(j);outbox.appendRow([j.key,j.target,raw,dispatchBridgeSign_(raw),'','']);});return {ok:true,status:'queued',executed:false,message:'Đã gửi app Kho, chưa ghi sổ. Xem kết quả từ Kho.',externalKeys:jobs.map(function(j){return j.key;})};}
function aiReadStockBook_(id){var book=DispatchIntegration.verified(SpreadsheetApp.openById.bind(SpreadsheetApp),id,PropertiesService.getScriptProperties().getProperty('DISPATCH_COPY_TOKEN'));var metadata=dispatchBridgeBook_('aiMetadata');var stockNames=['TONKHO','CHOTHUE','NHAT_KY_XNT','NHAT_KY_XNT_BACKUP','BACKUP_NHAT_KY_XNT'];
 function readOnly(object){return new Proxy(object,{get:function(target,name){var value=target[name];if(typeof value!=='function')return value;return function(){if(/^(set|clear|delete|insert|append|copy|move|sort|remove|protect|hide|show|autoResize|merge|unmerge)/.test(String(name)))throw new Error('KHO_SINGLE_WRITER');var result=value.apply(target,arguments);if(Array.isArray(result))return result.map(function(x){return x&&typeof x==='object'&&(typeof x.getRange==='function'||typeof x.getValues==='function'||typeof x.getSheets==='function')?readOnly(x):x;});return result&&typeof result==='object'&&(typeof result.getRange==='function'||typeof result.getValues==='function'||typeof result.getSheets==='function')?readOnly(result):result;};},set:function(){throw new Error('KHO_SINGLE_WRITER');}});}
 return new Proxy(book,{get:function(target,name){if(name==='getSheetByName')return function(sheetName){if(stockNames.indexOf(sheetName)>=0){var sheet=target.getSheetByName(sheetName);return sheet?readOnly(sheet):null;}return metadata.getSheetByName(sheetName);};if(name==='insertSheet')return function(sheetName){if(stockNames.indexOf(sheetName)>=0)throw new Error('KHO_SINGLE_WRITER');return metadata.insertSheet(sheetName);};return readOnly(target)[name];},set:function(){throw new Error('KHO_SINGLE_WRITER');}});
}
