function renderFinanceClosings() {
 if(!state.data||!['ADMIN','KE_TOAN'].includes(state.data.user.role))return;
 document.getElementById('closingDate').value=document.getElementById('closingDate').value||state.data.today;
 runApi('listFinanceClosings',[],function(result){document.getElementById('financeClosingRows').innerHTML='<p>'+ (result.enabled_at?'Đã bật lại chốt sổ':'Chưa bật lại chốt sổ')+'</p><table><thead><tr><th>Ngày</th><th>Trạng thái</th><th>Thực đếm</th><th>Chênh lệch</th><th>Mở khóa</th></tr></thead><tbody>'+result.rows.map(function(r){return '<tr><td>'+escapeHtml(r.date)+'</td><td>'+escapeHtml(r.legacy?'Dòng cũ · không áp dụng khóa':r.locked?'Đã khóa':r.status)+'</td><td>'+money(r.actual_cash)+'</td><td>'+money(r.difference)+'</td><td>'+(r.locked&&state.data.user.role==='ADMIN'?'<button type="button" data-unlock-finance-day="'+escapeAttribute(r.id)+'">Mở khóa</button>':escapeHtml(r.unlocked_by))+'</td></tr>';}).join('')+'</tbody></table>';document.querySelectorAll('[data-unlock-finance-day]').forEach(function(button){button.onclick=function(){const reason=prompt('Lý do mở khóa ngày (bắt buộc):');if(!reason||!reason.trim())return;runApi('unlockFinanceDay',[{id:button.dataset.unlockFinanceDay,reason:reason}],function(r){showToast(r.message,'success');renderFinanceClosings();},handleError);};});},handleError);
}
function enableClosingGuardUi(){if(!confirm('Bật lại chốt sổ? Dòng chốt cũ không tự khóa lại.'))return;runApi('enableFinanceClosingGuard',[],function(){renderFinanceClosings();showToast('Đã bật lại chốt sổ, ngày cũ giữ mở.','success');},handleError);}
