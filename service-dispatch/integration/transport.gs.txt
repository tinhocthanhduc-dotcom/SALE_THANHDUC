/** Shared ONLY by integration copies. No public anonymous endpoint and no destination sheet writes. */
function dispatchBridgeOwner_(){var a=Session.getActiveUser().getEmail(),b=Session.getEffectiveUser().getEmail();if(!a||a!==b)throw new Error('OWNER_SETUP_REQUIRED');}
function dispatchBridgeCopyId_(kind){var p=PropertiesService.getScriptProperties(),id=p.getProperty('DISPATCH_COPY_'+kind.toUpperCase()+'_ID');if(!id||DispatchIntegration.LEGACY_IDS.indexOf(id)>=0)throw new Error('INTEGRATION_COPY_REQUIRED');return id;}
function dispatchBridgeBook_(kind){return DispatchIntegration.verified(SpreadsheetApp.openById.bind(SpreadsheetApp),dispatchBridgeCopyId_(kind),PropertiesService.getScriptProperties().getProperty('DISPATCH_COPY_TOKEN'));}
function dispatchBridgeSign_(raw){var key=PropertiesService.getScriptProperties().getProperty('DISPATCH_BRIDGE_HMAC');if(!key)throw new Error('INTEGRATION_SECRET_REQUIRED');return Utilities.base64Encode(Utilities.computeHmacSha256Signature(raw,key));}
function dispatchBridgeVerify_(raw,signature){var expected=dispatchBridgeSign_(raw),diff=expected.length^String(signature||'').length;for(var i=0;i<expected.length;i++)diff|=expected.charCodeAt(i)^(String(signature||'').charCodeAt(i)||0);if(diff)throw new Error('INTEGRATION_SIGNATURE_INVALID');}
function installDispatchConsumer(){dispatchBridgeOwner_();dispatchBridgeGuard_();var existing=ScriptApp.getProjectTriggers().filter(function(t){return t.getHandlerFunction()==='dispatchConsumerTick_';});if(!existing.length)ScriptApp.newTrigger('dispatchConsumerTick_').timeBased().everyMinutes(5).create();return {installed:true,minutes:5};}
/** Manual Run is the 'Nhận từ Điều phối' button/menu target. */
function receiveFromDispatch(){if(!Session.getActiveUser().getEmail())throw new Error('AUTH_REQUIRED');return dispatchReceiveFromDispatch_();}
function dispatchConsumerTick_(event){if(!event||!ScriptApp.getProjectTriggers().some(function(t){return t.getUniqueId()===String(event.triggerUid)&&t.getHandlerFunction()==='dispatchConsumerTick_';}))throw new Error('AUTH_REQUIRED');return dispatchReceiveFromDispatch_();}
function dispatchReceiveFromDispatch_(){dispatchBridgeGuard_();var source=dispatchBridgeBook_('outbox'),sheet=source.getSheetByName('DISPATCH_OUTBOX');if(!sheet)throw new Error('INTEGRATION_OUTBOX_REQUIRED');var results=[],rows=sheet.getLastRow()>1?sheet.getRange(2,1,sheet.getLastRow()-1,6).getValues():[];rows.forEach(function(row,index){if(!row[0]||row[1]!==DISPATCH_CONSUMER_TARGET||row[4])return;try{dispatchBridgeVerify_(row[2],row[3]);var job=JSON.parse(row[2]);if(job.key!==row[0]||job.target!==row[1])throw new Error('INVALID_SYNC_DOCUMENT');var result=dispatchBridgeConsume_(job),report={key:job.key,attempt:job.attempt,ok:true,result:result};}catch(error){var report={key:row[0],attempt:typeof job==='object'&&job?job.attempt:0,ok:false,reason:String(error.message||error).slice(0,300)};}var current=sheet.getRange(index+2,1,1,6).getValues()[0];if(current[2]!==row[2]||current[3]!==row[3])return;var raw=JSON.stringify(report);sheet.getRange(index+2,5,1,2).setValues([[raw,dispatchBridgeSign_(raw)]]);results.push({key:row[0],ok:report.ok});});return {processed:results.length,results:results};}
