async function compressPhoto(file){
 const started=performance.now(),image=await createImageBitmap(file),canvas=document.createElement('canvas'),factor=Math.min(1,1280/Math.max(image.width,image.height));canvas.width=Math.max(1,Math.round(image.width*factor));canvas.height=Math.max(1,Math.round(image.height*factor));canvas.getContext('2d').drawImage(image,0,0,canvas.width,canvas.height);
 const thumb=document.createElement('canvas'),small=Math.min(1,320/Math.max(image.width,image.height));thumb.width=Math.max(1,Math.round(image.width*small));thumb.height=Math.max(1,Math.round(image.height*small));thumb.getContext('2d').drawImage(image,0,0,thumb.width,thumb.height);image.close();
 const encode=(surface,limit,qualities)=>{for(const quality of qualities){const raw=surface.toDataURL('image/jpeg',quality).split(',')[1],bytes=Math.floor(raw.length*3/4)-(raw.endsWith('==')?2:raw.endsWith('=')?1:0);if(bytes<=limit)return {raw,bytes,quality};}throw Error('INVALID_IMAGE');};
 const full=encode(canvas,199999,[.7,.6,.5,.4,.3,.2,.1]),mini=encode(thumb,30000,[.65,.5,.35,.2,.1]);photoTiming({compression_ms:performance.now()-started,bytes:full.bytes,thumbnail_bytes:mini.bytes,quality:full.quality,long_edge:Math.max(canvas.width,canvas.height)});return {base64:full.raw,thumbnail_base64:mini.raw};
}
