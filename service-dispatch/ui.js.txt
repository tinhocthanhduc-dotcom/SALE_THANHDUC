async function api(input){

function resetLogin(){data=null;selected=null;page='board';document.getElementById('app').hidden=true;document.getElementById('login').hidden=false;document.getElementById('user').textContent='';sessionStorage.removeItem('dispatch-token');}

const error=err=>message(errors[err.message]||'Yêu cầu chưa được hoàn tất: '+err.message,true);
function schedule(o){return [...o.appointments].reverse().find(a=>a.kind==='confirmed')||o.appointments.at(-1);}
function late(o){return !['working','accepted_work','cancelled'].includes(o.status)&&Date.now()>Date.parse(schedule(o).end);}
function board(){const orders=data.orders.filter(o=>page!=='mine'||o.technician_id===data.actor.id);return `<div class="stats">${[['Đơn đang mở',orders.filter(o=>!['accepted_work','cancelled'].includes(o.status)).length],['Chờ linh kiện',orders.filter(o=>o.status==='waiting_parts').length],['Cần điều phối xử lý',orders.filter(o=>o.transfers.some(t=>t.status==='pending')||o.status==='unable'||late(o)).length]].map(([title,n])=>`<div>${e(title)}<strong>${n}</strong></div>`).join('')}</div><h2 style="margin-top:20px">${page==='mine'?'Việc của tôi':'Bảng điều phối'}</h2><div class="grid">${orders.sort((a,b)=>Date.parse(schedule(a).start)-Date.parse(schedule(b).start)).map(o=>`<article class="card ${late(o)?'late':''}"><small>${e(o.order_no)}</small><h2>${e(o.customer_snapshot.name)}</h2><p>${e(o.machine_snapshot.model)} · ${e(o.request)}</p><p>${e(o.address)} · ${e(o.area)}</p><p>${fmt(schedule(o).start)} → ${fmt(schedule(o).end)}</p><span class="pill">${e(names[o.status])}</span><p>${e(data.technicians.find(t=>t.id===o.technician_id)?.name||'Chưa giao thợ')}${late(o)?' · Trễ hẹn':''}</p><button data-open="${e(o.id)}">Mở đơn</button></article>`).join('')||'<div class="panel">Chưa có đơn trong phạm vi của bạn.</div>'}</div>`;}
function customers(){return `<div class="panel"><h2>Khách hàng và hồ sơ máy</h2><div class="actions"><button data-form="create_customer">+ Khách hàng</button><button data-form="create_machine">+ Hồ sơ máy</button></div>${data.customers.map(c=>`<p><strong>${e(c.name)}</strong> · ${e(c.phone)} · ${e(c.address)} · ${e(c.terms==='credit'?'Công nợ':'Thu ngay')}</p>`).join('')||'<p>Chưa có khách mẫu.</p>'}</div>`;}
function detail(){const o=data.orders.find(o=>o.id===selected);if(!o)return '';const buttons=[];const b=type=>buttons.push(`<button class="${['accept','schedule','start','complete'].includes(type)?'primary':''}" data-form="${type}">${e(labels[type])}</button>`);
 if(manage()&&['new','assigned','accepted','scheduled','unable','waiting_parts'].includes(o.status))b('assign');
 if(has('technician')&&o.technician_id===data.actor.id){if(o.status==='assigned')b('accept');if(['accepted','scheduled','unable'].includes(o.status))b('schedule');if(o.status==='scheduled')b('start');if(o.status==='working'){b('wait_parts');b('cannot_complete');b('complete');}if(o.status==='scheduled')b('cannot_complete');if(['accepted','scheduled','working','waiting_parts','unable'].includes(o.status)&&!o.transfers.some(t=>t.status==='pending'))b('request_transfer');}
 if(manage()&&o.transfers.some(t=>t.status==='pending')){b('approve_transfer');b('reject_transfer');}
 if(manage()&&!['accepted_work','cancelled'].includes(o.status))b('cancel');
 return `<div class="panel" style="margin-top:20px"><h2>${e(o.order_no)} · ${e(names[o.status])}</h2><div class="facts"><div><strong>${e(o.customer_snapshot.name)}</strong><p>${e(o.customer_snapshot.phone)}</p><p>${e(o.address)}</p><p>${e(o.machine_snapshot.model)} · ${e(o.machine_snapshot.serial)}</p></div><div><p>${e(o.request)}</p><p>Hẹn: ${fmt(schedule(o).start)} → ${fmt(schedule(o).end)}</p><p>Thanh toán: chưa triển khai · ${e(o.terms==='credit'?'Công nợ – không thu tại chỗ':'Thu ngay')}</p><p>Phiên bản ${o.version}</p></div></div>${o.unable_reason?`<p class="error">Không làm được: ${e(o.unable_reason)}</p>`:''}${o.work_summary?`<p>Đã nghiệm thu: ${e(o.work_summary)} · Hồ sơ chờ đối soát</p>`:''}<div class="actions">${buttons.join('')}<button data-copy="${e(o.id)}">Sao chép link gửi Zalo</button></div>${o.transfers.map(t=>`<p>Chuyển thợ: ${e(t.reason)} · ${e(t.status)}</p>`).join('')}${o.parts_requests.map(r=>`<p>Vật tư: ${e(r.item)} · SL ${r.qty} · Chờ kết nối kho xác nhận</p>`).join('')}${o.status==='working'&&has('technician')&&o.technician_id===data.actor.id?`<form id="upload-form"><h3>Ảnh nghiệm thu</h3><label for="kind">Loại ảnh</label><select id="kind"><option value="after">Ảnh sau khi làm</option><option value="handover">Biên bản / chữ ký khách</option></select><label for="image-file">Chọn/chụp ảnh</label><input id="image-file" type="file" accept="image/jpeg,image/png,image/webp" capture="environment" required><div class="actions"><button>Tải ảnh lên</button></div></form>`:''}<div class="actions">${o.evidence.map(img=>`<button data-image="${e(img.id)}">${img.kind==='after'?'Ảnh sau':'Biên bản/chữ ký'}</button>`).join('')}</div><div id="image-view"></div><h3>Lịch sử</h3><div class="timeline">${data.events.filter(ev=>ev.order_id===o.id).map(ev=>`<p>${fmt(ev.at)} · ${e(labels[ev.type]||ev.type)} · ${e(ev.actor_id)} · v${ev.version}${ev.reason?' · '+e(ev.reason):''}</p>`).join('')}</div></div>`;

async function submitForm(form){if(busy)return;busy=true;form.querySelector('button').disabled=true;try{const payload=Object.fromEntries(new FormData(form));for(const key of ['appointment_start','appointment_end','start','end'])if(payload[key])payload[key]=new Date(payload[key]+':00+07:00').toISOString();if(payload.qty)payload.qty=Number(payload.qty);if(!form.dataset.type.startsWith('create_'))payload.order_id=selected;const result=await api({action:'command',command:{type:form.dataset.type,idempotency_key:form.dataset.key,expected_version:form.dataset.version?Number(form.dataset.version):undefined,payload}});if(result.order_id)selected=result.order_id;document.getElementById('dialog').close();await refresh();message('Đã lưu'+(result.order_no?' · '+result.order_no:'')+(result.warning==='DUPLICATE_PHONE'?' · Số điện thoại trùng khách khác; cần điều phối kiểm tra, chưa gộp hồ sơ.':''));}catch(err){document.getElementById('form-error').textContent=errors[err.message]||err.message;if(err.message==='AUTH_REQUIRED'){resetLogin();document.getElementById('dialog').close();}}finally{busy=false;form.querySelector('button').disabled=false;}}
async function upload(form){if(busy)return;busy=true;try{const o=data.orders.find(o=>o.id===selected),file=document.getElementById('image-file').files[0];if(!file)throw Error('INVALID_IMAGE');const image=await createImageBitmap(file);const canvas=document.createElement('canvas');const factor=Math.min(1,1600/Math.max(image.width,image.height));canvas.width=Math.round(image.width*factor);canvas.height=Math.round(image.height*factor);canvas.getContext('2d').drawImage(image,0,0,canvas.width,canvas.height);image.close();let raw;for(const quality of [.8,.65,.5,.35]){raw=canvas.toDataURL('image/jpeg',quality).split(',')[1];if(raw.length<=530000)break;}if(raw.length>530000)throw Error('INVALID_IMAGE');const kind=document.getElementById('kind').value,identity=[o.id,kind,file.name,file.size,file.lastModified].join(':');if(form.dataset.uploadIdentity!==identity){form.dataset.uploadIdentity=identity;form.dataset.uploadKey=uid();form.dataset.uploadVersion=String(o.version);}await api({action:'upload',payload:{order_id:o.id,expected_version:Number(form.dataset.uploadVersion),idempotency_key:form.dataset.uploadKey,kind,mime:'image/jpeg',base64:raw}});await refresh();message('Ảnh đã được máy chủ lưu.');}catch(err){error(err);}finally{busy=false;}}
