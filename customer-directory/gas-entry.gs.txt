function cdLock_(fn) {
  var l = LockService.getScriptLock();
  l.waitLock(2e4);
  try {
    return fn();
  } finally {
    l.releaseLock();
  }
}
function cdTryReadLock_(fn) {
  var l = LockService.getScriptLock();
  if (!l.tryLock(0)) return null;
  try { return fn(); } finally { l.releaseLock(); }
}
function cdHash_(s) {
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, s, Utilities.Charset.UTF_8).map(function(b) {
    return ("0" + ((b + 256) % 256).toString(16)).slice(-2);
  }).join("");
}
function cdHmac_(s, key) {
  return Utilities.computeHmacSha256Signature(s, key, Utilities.Charset.UTF_8).map(function(b) {
    return ("0" + ((b + 256) % 256).toString(16)).slice(-2);
  }).join("");
}
function cdWriteRows_(sheet, start, rows) {
  if (!rows.length) return;
  var end = start + rows.length - 1;
  if (end > sheet.getMaxRows()) sheet.insertRowsAfter(sheet.getMaxRows(), end - sheet.getMaxRows());
  if (rows[0].length > sheet.getMaxColumns()) sheet.insertColumnsAfter(sheet.getMaxColumns(), rows[0].length - sheet.getMaxColumns());
  sheet.getRange(start, 1, rows.length, rows[0].length).setRichTextValues(rows.map(function(row) {
    return row.map(function(v) {
      return SpreadsheetApp.newRichTextValue().setText(String(v === void 0 || v === null ? "" : v)).build();
    });
  }));
}

function cdReadCopies_(config) {
  var sources = JSON.parse(config.s.CD_SOURCES || "{}"), out = [];
  if (!config.s.CD_SOURCES_SIGNATURE || !DirectoryBundle.Auth.equal(config.s.CD_SOURCES_SIGNATURE, cdHmac_(DirectoryBundle.stable(sources), config.s.CD_MARKER))) DirectoryBundle.Core.fail("COPY_SOURCES_REQUIRED");
  ["SALE", "CARE", "FIN", "DISPATCH"].forEach(function(app) {
    var id = sources[app];
    if (!id || DirectoryBundle.ORIGINAL_IDS.indexOf(id) >= 0 || id === config.s.CD_WORKBOOK_ID) DirectoryBundle.Core.fail("COPY_SOURCES_REQUIRED");
    var book = SpreadsheetApp.openById(id);
    if (app === "DISPATCH") {
      var state = cdReadTable_(book, "ERP_STATE"), journal = cdReadTable_(book, "ERP_JOURNAL"), snapshot = DirectoryBundle.Sources.decodeSnapshot(state.rows, cdHash_, ["SNAPSHOT"]), intent = DirectoryBundle.Sources.decodeSnapshot(journal.rows, cdHash_, ["PREPARED", "COMMITTED"]);
      if (intent && (!snapshot || intent.revision > snapshot.revision)) DirectoryBundle.Core.fail("DISPATCH_COPY_UNSETTLED");
      out = out.concat(DirectoryBundle.Sources.fromDispatch(snapshot ? snapshot.state : { customers: {}, orders: {} }));
    } else {
      var tables = {};
      tables[DirectoryBundle.Sources.SOURCE_TABLES[app]] = cdReadTable_(book, DirectoryBundle.Sources.SOURCE_TABLES[app]);
      if (app === "FIN") tables.CONG_NO_KH = cdReadTable_(book, "CONG_NO_KH");
      out = out.concat(DirectoryBundle.Sources.fromTables(app, tables));
    }
  });
  return out;
}
